27. 更新 PHP 加载文件 ▶

PHP 修改方案

// 方法 A: wp_script_add_data（推荐）
wp_enqueue_script('pwca-boundary', ..., [], '1.0', true);
wp_script_add_data('pwca-boundary', 'type', 'module');

// 方法 B: filter_script_loader_tag（批量）
add_filter('script_loader_tag', function($tag, $handle) {
$module_handles = ['pwca-boundary', 'pwca-canvas-manager', ...];
if (in_array($handle, $module_handles)) {
return str_replace("type='text/javascript'", "type='module'", $tag);
}
return $tag;
}, 10, 2);
